┌─────────────────────────────────────────────────────────────────────┐

    counters.service_type_id ──────────> service_types.id
        ON DELETE: SET NULL
        
    tickets.service_type_id ───────────> service_types.id
        ON DELETE: CASCADE
        
    tickets.counter_id ────────────────> counters.id

┌─────────────────────────────────────────────────────────────────────┐
│                      DATA INTEGRITY RULES                            │
└─────────────────────────────────────────────────────────────────────┘

    ✅ UNIQUE Constraints:
       - service_types.name
       - service_types.code
       - counters.counter_number
       - tickets.ticket_number
       
    ✅ CHECK Constraints:
       - tickets.status IN ('WAITING', 'SERVING', 'COMPLETED', 'CANCELLED')
       
    ✅ NOT NULL Constraints:
       - All primary keys
       - service_types.name, service_types.code
       - counters.counter_number, counters.name
       - tickets.ticket_number, tickets.service_type_id
       - tickets.status, tickets.queue_position
       
    ✅ DEFAULT Values:
       - is_active = 1 (TRUE)
       - timestamps = CURRENT_TIMESTAMP
       - average_service_time = 10 minutes


┌─────────────────────────────────────────────────────────────────────┐
│                      SAMPLE QUERY PATTERNS                           │
└─────────────────────────────────────────────────────────────────────┘

    1️⃣  Get next waiting ticket for a service:
        SELECT * FROM tickets 
        WHERE service_type_id = ? AND status = 'WAITING'
        ORDER BY queue_position ASC LIMIT 1;
        
    2️⃣  Get queue statistics:
        SELECT * FROM queue_statistics;
        
    3️⃣  Get active queue for display:
        SELECT * FROM active_queue 
        WHERE service_type_id = ?;
        
    4️⃣  Issue new ticket:
        INSERT INTO tickets 
        (ticket_number, service_type_id, status, queue_position)
        VALUES (?, ?, 'WAITING', ?);
        
    5️⃣  Call ticket to counter:
        UPDATE tickets 
        SET status = 'SERVING', 
            counter_id = ?,
            called_at = CURRENT_TIMESTAMP
        WHERE id = ?;
        
    6️⃣  Complete service:
        UPDATE tickets 
        SET status = 'COMPLETED',
            completed_at = CURRENT_TIMESTAMP
        WHERE id = ?;
